category: Mains
time: 1 hr
servings: 6
difficulty: 2
ingredients:
  - name: Potatoes, thinly sliced
    amount: 5
    unit: "pieces"
  - name: Onion, chopped
    amount: 1
    unit: "piece"
  - name: Curry powder
    amount: 1
    unit: "tsp"
  - name: Origanum (oregano)
    amount: 1
    unit: "tsp"
  - name: Garlic flakes
    amount: 1
    unit: "tsp"
  - name: Beef mince
    amount: 500
    unit: "g"
  - name: Tomato paste
    amount: 50
    unit: "g"
  - name: Beef stock cubes, dissolved in 1 cup hot water
    amount: 2
    unit: "pieces"
  - name: "White sauce: butter"
    amount: 80
    unit: "g"
  - name: "White sauce: flour"
    amount: 80
    unit: "g"
  - name: "White sauce: milk"
    amount: 3
    unit: "cups"
  - name: "White sauce: nutmeg"
    amount: 0
    unit: "pinch"
  - name: "Crust: oil"
    amount: 125
    unit: "g"
  - name: "Crust: milk"
    amount: 125
    unit: "g"
  - name: "Crust: eggs"
    amount: 2
    unit: "pieces"
  - name: "Crust: self-raising flour"
    amount: 1
    unit: "cup"
  - name: Salt & pepper
    amount: 0
    unit: "to taste"
---

<div class="step">
  <div class="step-num">1</div>
  <p class="step-text"><strong>Prepare the potatoes.</strong> Thinly slice and microwave for about <strong>10 minutes</strong>, until slightly softened. Set aside.</p>
</div>
<div class="step">
  <div class="step-num">2</div>
  <p class="step-text"><strong>Make the filling.</strong> Sauté the onion until soft. Add the curry powder, origanum, garlic flakes, salt and pepper, then the mince, and cook until browned. Stir in the tomato paste, pour in the beef stock and simmer for 10&ndash;15 minutes until slightly thickened.</p>
</div>
<div class="step">
  <div class="step-num">3</div>
  <p class="step-text"><strong>Make the white sauce.</strong> Melt the butter, stir in the flour and cook for 1 minute. Gradually whisk in the milk until smooth, simmer until thickened, and season with salt, pepper and a pinch of nutmeg.</p>
</div>
<div class="step">
  <div class="step-num">4</div>
  <p class="step-text"><strong>Make the crust.</strong> Whisk together the oil, milk and eggs, season, then fold in the self-raising flour until smooth.</p>
</div>
<div class="step">
  <div class="step-num">5</div>
  <p class="step-text"><strong>Assemble.</strong> In a greased baking dish, layer the potato slices, then the mince filling, then the white sauce, and spread the crust mixture over the top.</p>
</div>
<div class="step">
  <div class="step-num">6</div>
  <p class="step-text">Bake at <strong>180°C for about 35 minutes</strong>, until golden brown and cooked through. Rest for 10 minutes before serving.</p>
</div>
