difficulty: 2
ingredients:
  - name: Beef mince
    amount: 550
    unit: "g"
  - name: Onion, finely chopped
    amount: 1
    unit: "large"
  - name: Mozzarella
    amount: 125
    unit: "g"
  - name: Canned chopped tomatoes
    amount: 400
    unit: "g"
  - name: Breadcrumbs
    amount: 0.5
    unit: "cup"
  - name: Milk
    amount: 4
    unit: "tbsp"
  - name: Dijon mustard
    amount: 1
    unit: "tbsp"
  - name: Cumin
    amount: 1
    unit: "tsp"
  - name: Fresh basil
    amount: 0
    unit: "a handful"
  - name: Salt & pepper
    amount: 0
    unit: "to taste"
---

<div class="step">
  <div class="step-num">1</div>
  <p class="step-text">Fry the chopped onion gently with the dijon mustard until soft and golden.</p>
</div>
<div class="step">
  <div class="step-num">2</div>
  <p class="step-text">In a bowl, soak the breadcrumbs in the milk, then mix in the mince, fried onion, cumin, salt, pepper and any other spices. Roll into balls.</p>
</div>
<div class="step">
  <div class="step-num">3</div>
  <p class="step-text">Sear the meatballs in a hot pan until browned all over.</p>
</div>
<div class="step">
  <div class="step-num">4</div>
  <p class="step-text">Add the mozzarella, basil and chopped tomatoes, then bake for <strong>25 minutes at 180&deg;C</strong>.</p>
</div>
