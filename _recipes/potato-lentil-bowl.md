ingredients:
  - name: Onion
    amount: 1
    unit: "medium"
  - name: Potatoes
    amount: 2
    unit: "medium"
  - name: Lentils
    amount: 125
    unit: "g"
  - name: Mixed veg
    amount: 0
    unit: "any"
  - name: Stock
    amount: 1
    unit: "litre"
  - name: Leafy greens
    amount: 0
    unit: "a handful"
  - name: Salt & pepper
    amount: 0
    unit: "to taste"
---

<div class="step">
  <div class="step-num">1</div>
  <p class="step-text">Add the onion, potatoes, veg and spices to a pot.</p>
</div>
<div class="step">
  <div class="step-num">2</div>
  <p class="step-text">Pour in 1 litre of water with the stock.</p>
</div>
<div class="step">
  <div class="step-num">3</div>
  <p class="step-text">After <strong>30 minutes</strong>, add the lentils.</p>
</div>
<div class="step">
  <div class="step-num">4</div>
  <p class="step-text">Ten minutes later add the leafy greens and remove the lid.</p>
</div>
<div class="step">
  <div class="step-num">5</div>
  <p class="step-text">Serve once the liquid has cooked down.</p>
</div>
