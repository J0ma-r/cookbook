category: Mains
tags: [Warm]
time: 40 min
servings: 3
difficulty: 2
ingredients:
  - name: Unprocessed sausage
    amount: 4
    unit: "pieces"
  - name: Rice
    amount: 0.5
    unit: "cup"
  - name: Purple onion, chopped
    amount: 1
    unit: "medium"
  - name: Garlic
    amount: 2
    unit: "cloves"
  - name: Red pepper, chopped
    amount: 1
    unit: "piece"
  - name: Chicken stock
    amount: 1
    unit: "cup"
  - name: White wine
    amount: 0
    unit: "optional"
  - name: Fresh tomatoes, chives, parsley
    amount: 0
    unit: "to finish"
---

<div class="step">
  <div class="step-num">1</div>
  <p class="step-text">Fry the chopped onion in oil in a pot.</p>
</div>
<div class="step">
  <div class="step-num">2</div>
  <p class="step-text">Add the chopped red pepper and garlic.</p>
</div>
<div class="step">
  <div class="step-num">3</div>
  <p class="step-text">Squeeze the sausage meat out of its skins and add in chunks over high heat.</p>
</div>
<div class="step">
  <div class="step-num">4</div>
  <p class="step-text">Stir in the rice, the wine if using, and the chicken stock.</p>
</div>
<div class="step">
  <div class="step-num">5</div>
  <p class="step-text">Cook until thick, then finish with fresh tomatoes and herbs.</p>
</div>
