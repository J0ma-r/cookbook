ingredients:
  - name: Chicken breasts
    amount: 2
    unit: "pieces"
  - name: Fresh basil
    amount: 0
    unit: "a few leaves"
  - name: Sundried tomatoes
    amount: 6
    unit: "pieces"
  - name: Chorizo or cured meat
    amount: 60
    unit: "g"
  - name: Mozzarella
    amount: 80
    unit: "g"
---

<div class="step">
  <div class="step-num">1</div>
  <p class="step-text">Slice each chicken breast in half horizontally to make a pocket.</p>
</div>
<div class="step">
  <div class="step-num">2</div>
  <p class="step-text">Stuff with as much chorizo, sundried tomato, basil and mozzarella as will fit.</p>
</div>
<div class="step">
  <div class="step-num">3</div>
  <p class="step-text">Sear the outside of the chicken in a hot pan to colour it.</p>
</div>
<div class="step">
  <div class="step-num">4</div>
  <p class="step-text">Bake for <strong>30&ndash;35 minutes at 180&deg;C</strong>. Serve with a pepper or cheese sauce if you like.</p>
</div>
